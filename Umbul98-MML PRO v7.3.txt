N = nVar2

if N <= 0
    N := 1.0

     10.0,
     math.ceil(math.log(safeRange / N) / math.log(10.0)))

    SI := syminfo.mintick

Top = I + SI

    Top += SI

while Bot > vLow
    Bot -= SI

getSignal(level) =>
    signal = "NÖTR"

    touched = low <= level + touchDistance and high >= level - touchDistance

    bullishBreak = ta.crossover(close, level) and close > level + breakDistance
    bearishBreak = ta.crossunder(close, level) and close < level - breakDistance

    bullishReversal = touched and close > level and (not requireBodyConfirmation or bullCandle) and (not requireCloseConfirmation or close > level)
    bearishReversal = touched and close < level and (not requireBodyConfirmation or bearCandle) and (not requireCloseConfirmation or close < level)

    support = touched and low <= level + touchDistance and close >= level
    resistance = touched and high >= level - touchDistance and close <= level

    if enableSignalEngine
        if bullishBreak
            signal := "KIRILIM ↑"
        else if bearishBreak
            signal := "KIRILIM ↓"
        else if bullishReversal
            signal := "DÖNÜŞ ↑"
        else if bearishReversal
            signal := "DÖNÜŞ ↓"
        else if support
            signal := "DESTEK"
        else if resistance
            signal := "DİRENÇ"

    signal
